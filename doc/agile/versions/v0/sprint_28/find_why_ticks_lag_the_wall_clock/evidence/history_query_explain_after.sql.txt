    FROM ores_marketdata_ticks_tbl t
    WHERE t.series_id = 'cccccccc-0000-0000-0000-000000000001'::uuid
      AND t.ts >= (now() - interval '30 days') AND t.ts < now()
    ORDER BY t.ts
    LIMIT 25001
)
SELECT (extract(epoch FROM p.ts) * 1000000)::bigint::text, c.oresmd_uri, p.value::text
FROM page p
JOIN ores_marketdata_tick_coords_tbl c
  ON c.series_id = 'cccccccc-0000-0000-0000-000000000001'::uuid
 AND c.coord = p.coord AND c.party_id = p.party_id
ORDER BY p.ts, p.coord;
