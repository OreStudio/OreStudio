The message functions and =ores_outcome_raise_fn=, from the outcome catalogue.
=outcome_catalogue= profile. One function per store outcome composes that
outcome's sentence; the raiser maps an outcome code to the SQLSTATE the trigger
must raise and calls the function. A trigger names a code and a field and
carries no wording of its own.

-- Fills a message template's {name} placeholders with the matching values. The
-- one renderer, shared by every outcome function below and mirrored by
-- ores::database::domain::describe in C++.
create or replace function ores_outcome_fill_fn(
    p_template text,
    p_names text[],
    p_values text[]
) returns text language plpgsql immutable as $$
declare
    v_text text := p_template;
    i integer;
begin
    for i in 1 .. coalesce(array_length(p_names, 1), 0) loop
        v_text := replace(v_text, '{' || p_names[i] || '}', coalesce(p_values[i], ''));
    end loop;
    return v_text;
end;
$$;
