from pathlib import Path


def edit(path, old, new):
    p = Path(path)
    t = p.read_text()
    assert t.count(old) == 1, (path, old)
    p.write_text(t.replace(old, new))


def append(path, text):
    p = Path(path)
    p.write_text(p.read_text() + text)
