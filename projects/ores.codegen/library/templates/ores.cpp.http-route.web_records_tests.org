The tests of one resource's record routes. Each case runs a handler against a
client that is not connected, so it needs no server, no NATS and no database.
A request the route refuses comes back with the BFF's status and its
={code, message}= body. A request the route accepts fails at the forward step,
which the route answers as 502 =internal=, so a case can tell the two apart.

Each case name starts with the unit's stem, because a component's test files
link into one binary and Catch2 refuses a duplicate name.

/// Runs a handler to completion. The handler never reaches a socket.
http_response {{unit}}_run(boost::asio::awaitable<http_response> work) {
    boost::asio::io_context io;
    std::optional<http_response> answer;
    boost::asio::co_spawn(
        io,
        [&answer, work = std::move(work)]() mutable -> boost::asio::awaitable<void> {
            answer = co_await std::move(work);
        },
        boost::asio::detached);
    io.run();
    return answer.value_or(http_response{});
}

/// A request with these query parameters and this body.
http_request {{unit}}_request(std::initializer_list<std::pair<const std::string, std::string>> params,
                              std::string body = {}) {
    http_request req;
    req.query_params = params;
    req.body = std::move(body);
    return req;
}

bool {{unit}}_says(const http_response& response, const std::string& code) {
    return response.body.find("\"code\":\"" + code + "\"") != std::string::npos;
}
