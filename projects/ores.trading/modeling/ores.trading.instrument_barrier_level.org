One level of a barrier, keyed to the barrier and the level's ordinal.

A single barrier states one level and a double barrier states two; the
levels are a list in the document, so each is a row of its own rather
than a column of the barrier.

** Custom repository methods

The trade read path fetches the instrument barrier levels of many instruments at
once, and no generated read covers that: the generated reads address one key
group at a time. The batch read below is one statement over the instrument
set.

#+begin_src cpp :name implementation :implements F2EB1914-5E94-42CA-9C77-46BDB364BF9E
std::vector<domain::instrument_barrier_level>
instrument_barrier_level_repository::read_by_instruments_batch(
    context ctx, const std::vector<std::string>& trade_ids) {
    if (trade_ids.empty())
        return {};
    static constexpr auto max = db_timestamp::max();
    const auto tid = ctx.tenant_id().to_string();
    const auto query = query_builder::read<std::vector<instrument_barrier_level_entity>> |
                       query_builder::where("tenant_id"_c == tid &&
                             "trade_id"_c.in(trade_ids) && "valid_to"_c == max) |
                       query_builder::order_by("trade_id"_c, "barrier_role"_c, "barrier_sequence_number"_c, "sequence_number"_c);
    return query_builder::execute_read_query<instrument_barrier_level_entity,
                                             domain::instrument_barrier_level>(
        ctx,
        query,
        [](const auto& entities) { return instrument_barrier_level_mapper::map(entities); },
        lg(),
        "Reading instrument barrier levels for multiple instruments.");
}
