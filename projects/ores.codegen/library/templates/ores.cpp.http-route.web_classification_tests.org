The tests of one code list's routes. Each case runs a handler against a client
that is not connected, so it needs no server, no NATS and no database. A
request the route refuses comes back with the BFF's status and its
={code, message}= body; a request the route accepts fails at the forward step
as the service not answering.

http_response {{unit}}_run(boost::asio::awaitable<http_response> work) {
    boost::asio::io_context io;
    std::optional<http_response> answer;
    boost::asio::co_spawn(
        io,
        [&answer, work = std::move(work)]() mutable -> boost::asio::awaitable<void> {
            answer = co_await std::move(work);
        },
        boost::asio::detached);
    io.run();
    return answer.value_or(http_response{});
}

http_request {{unit}}_request(std::string body, std::string code = "X") {
    http_request req;
    req.body = std::move(body);
    req.path_params["code"] = std::move(code);
    return req;
}

TEST_CASE("{{unit}}_routes_registers_the_screen_addresses", "[routes][classification]") {
    nats_client session;
    auto table = std::make_shared<router>();
    {{unit}}_routes::register_routes(table, std::make_shared<endpoint_registry>(), session);
    const auto& routes = table->routes();
    const auto has = [&](http_method method, const std::string& pattern) {
        return std::any_of(routes.begin(), routes.end(), [&](const auto& route) {
            return route.method == method && route.pattern == pattern && route.requires_auth;
        });
    };
    CHECK(has(http_method::get, "{{base}}"));
    CHECK(has(http_method::post, "{{base}}/rows"));
    CHECK(has(http_method::put, "{{base}}/rows/{code}"));
    CHECK(has(http_method::put, "{{base}}/order"));
    CHECK(has(http_method::delete_, "{{base}}/rows/{code}"));
    CHECK(has(http_method::put, "{{base}}/rows/{code}/label"));
}
