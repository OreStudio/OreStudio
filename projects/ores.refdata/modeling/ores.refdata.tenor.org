#+created: 2026-07-13
#+updated: 2026-07-13

Persisted catalog of the standard [[id:0AC88EB3-DB7F-4135-9DA6-0ED4583FEC29][tenor]]
labels, so a user can see, manage, and order the set of tenors the system
knows about rather than relying on labels implied by parsing logic alone.
This table carries convention-independent identity only — how a given
tenor resolves under a particular curve type is
[[id:C4D8A2E6-3B7F-4A1D-9C5E-8F2A6D3B1E90][Tenor Convention]]'s concern,
recorded via [[id:E1F5A9C3-6D2B-4E8A-B7F1-3C9D5A2E6B48][Tenor Convention
Resolution]]. [[id:4DF1F49A-9AB4-4B56-B170-808490071799][Broken dates]]
are deliberately out of scope — they are per-instance ladder data, not
master data, and reference a row here only when they happen to coincide
with a standard label.

** Custom NATS protocol messages

=tenor= declares one message pair beyond the standard CRUD surface and the
generic facets, in the top-level =* Messages= section: =resolve_tenor=. The
header and its TypeScript twin both render it from there.

The handler is an adapter and decides nothing: it proves the request,
checks the permission a read needs, decodes the canonical request, calls
=tenor_resolution_service= and replies with the response the service
filled. The arithmetic is the pure engine's, not a second copy of it here.

#+begin_src cpp :name custom_handler_methods :implements B9E2F4A1-3C7D-4E8B-A5F0-2D6C1B8E3A7F
    /**
     * @brief Serves refdata.v1.ops.resolve_tenor.
     *
     * A refusal is a normal typed reply carrying a non-ok =result=, not a
     * thrown exception: a caller reads which step refused, the dates it had
     * counted to, and the row at fault, out of the payload. Only the
     * transport failing -- no session, no permission, an undecodable
     * request -- is an error reply.
     */
    void resolve_tenor(ores::nats::message msg) {
        BOOST_LOG_SEV(tenor_handler_lg(), debug) << "Handling " << msg.subject;
        auto req_ctx_expected = ores::service::service::make_request_context(ctx_, msg, verifier_);
        if (!req_ctx_expected) {
            error_reply(nats_, msg, req_ctx_expected.error());
            return;
        }
        const auto& req_ctx = *req_ctx_expected;
        [[maybe_unused]] const auto correlation_id = ores::service::messaging::log_handler_entry(
            tenor_handler_lg(), msg, req_ctx.tenant_id().to_string());
        if (!has_permission(req_ctx, "refdata::tenors:read")) {
            error_reply(nats_, msg, ores::service::error_code::forbidden);
            return;
        }
        auto req = decode<resolve_tenor_request>(msg);
        if (!req) {
            BOOST_LOG_SEV(tenor_handler_lg(), warn) << "Failed to decode: " << msg.subject;
            error_reply(nats_, msg, ores::service::error_code::bad_request);
            return;
        }
        service::tenor_resolution_service svc(req_ctx);
        try {
            auto response = svc.resolve(*req);
            BOOST_LOG_SEV(tenor_handler_lg(), debug) << "Completed " << msg.subject;
            reply(nats_, msg, response);
        } catch (const std::exception& e) {
            // The service reports what it decided in the response; an
            // exception here is the store failing, which is a different
            // thing and is reported as such.
            BOOST_LOG_SEV(tenor_handler_lg(), error) << msg.subject << " failed: " << e.what();
            resolve_tenor_response failure;
            failure.result.outcome = ores::utility::domain::outcome::failed;
            failure.result.code = "internal_error";
            failure.result.message = e.what();
            reply(nats_, msg, failure);
        }
    }

    /**
     * @brief Serves refdata.v1.ops.preview_tenor_resolution.
     *
     * The same question as resolve_tenor, asked about rows that are not
     * stored yet, so a person editing a rule reads its answer before writing
     * it. Same response, same refusals, same adapter: it proves the request,
     * checks the permission, decodes, calls the service and replies.
     */
    void preview_tenor_resolution(ores::nats::message msg) {
        BOOST_LOG_SEV(tenor_handler_lg(), debug) << "Handling " << msg.subject;
        auto req_ctx_expected = ores::service::service::make_request_context(ctx_, msg, verifier_);
        if (!req_ctx_expected) {
            error_reply(nats_, msg, req_ctx_expected.error());
            return;
        }
        const auto& req_ctx = *req_ctx_expected;
        [[maybe_unused]] const auto correlation_id = ores::service::messaging::log_handler_entry(
            tenor_handler_lg(), msg, req_ctx.tenant_id().to_string());
        if (!has_permission(req_ctx, "refdata::tenors:read")) {
            error_reply(nats_, msg, ores::service::error_code::forbidden);
            return;
        }
        auto req = decode<preview_tenor_resolution_request>(msg);
        if (!req) {
            BOOST_LOG_SEV(tenor_handler_lg(), warn) << "Failed to decode: " << msg.subject;
            error_reply(nats_, msg, ores::service::error_code::bad_request);
            return;
        }
        service::tenor_resolution_service svc(req_ctx);
        try {
            auto response = svc.preview(*req);
            BOOST_LOG_SEV(tenor_handler_lg(), debug) << "Completed " << msg.subject;
            reply(nats_, msg, response);
        } catch (const std::exception& e) {
            BOOST_LOG_SEV(tenor_handler_lg(), error) << msg.subject << " failed: " << e.what();
            resolve_tenor_response failure;
            failure.result.outcome = ores::utility::domain::outcome::failed;
            failure.result.code = "internal_error";
            failure.result.message = e.what();
            reply(nats_, msg, failure);
        }
    }
