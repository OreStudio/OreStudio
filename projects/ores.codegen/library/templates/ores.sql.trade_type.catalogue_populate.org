The trade types seed: one =insert ... values= row per trade type the
catalogue declares, in catalogue order. The script is idempotent, as the
hand-written one was. =description_sql= is the description folded onto one
line with its quotes doubled, so a description with an apostrophe cannot
break the literal.

 * Seeds every trade type ORE knows. Generated from

\echo '--- Trade Types ---'

insert into ores_trading_trade_types_tbl (
    code, tenant_id, version, description,
    product_type, has_options, has_extension,
    modified_by, change_reason_code, change_commentary
) values
{{#trade_type_catalogue}}
{{#rows}}
    ('{{code}}', ores_utility_system_tenant_id_fn(), 0, '{{description_sql}}',
     '{{product_type}}', {{has_options}}, {{has_extension}},
     'ores_trading_service', 'system.initial_load', 'Seed trade types'){{comma}}
{{/rows}}
{{/trade_type_catalogue}}
on conflict (tenant_id, code)
where valid_to = ores_utility_infinity_timestamp_fn()
do nothing;

select 'Trade Types' as entity, count(*) as count
from ores_trading_trade_types_tbl
where valid_to = ores_utility_infinity_timestamp_fn();
