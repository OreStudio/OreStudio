One tenor of a published FX forward set. The row holds every number the ladder
shows for the tenor, as the set computed them, so a reader gets the published
numbers back after the spot and the curves have moved.

std::vector<domain::fx_forward_publish_point>
fx_forward_publish_point_repository::read_latest_by_publish(context ctx, const boost::uuids::uuid& publish_id) {
    static constexpr auto max = db_timestamp::max();
    const auto tid = ctx.tenant_id().to_string();
    const auto pid = boost::uuids::to_string(publish_id);
    const auto query = query_builder::read<std::vector<fx_forward_publish_point_entity>> |
                       query_builder::where("tenant_id"_c == tid && "publish_id"_c == pid &&
                                            "valid_to"_c == max);
    return query_builder::execute_read_query<fx_forward_publish_point_entity, domain::fx_forward_publish_point>(
        ctx,
        query,
        [](const auto& entities) { return fx_forward_publish_point_mapper::map(entities); },
        lg(),
        "Reading the rows of one FX forward publish.");
}
