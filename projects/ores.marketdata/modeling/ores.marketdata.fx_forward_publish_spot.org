One spot observation a published FX forward set read. A USD pair reads one, its
own. A cross reads two, the USD leg of each currency, and derives its spot from
them, so the set names both.

std::vector<domain::fx_forward_publish_spot>
fx_forward_publish_spot_repository::read_latest_by_publish(context ctx, const boost::uuids::uuid& publish_id) {
    static constexpr auto max = db_timestamp::max();
    const auto tid = ctx.tenant_id().to_string();
    const auto pid = boost::uuids::to_string(publish_id);
    const auto query = query_builder::read<std::vector<fx_forward_publish_spot_entity>> |
                       query_builder::where("tenant_id"_c == tid && "publish_id"_c == pid &&
                                            "valid_to"_c == max);
    return query_builder::execute_read_query<fx_forward_publish_spot_entity, domain::fx_forward_publish_spot>(
        ctx,
        query,
        [](const auto& entities) { return fx_forward_publish_spot_mapper::map(entities); },
        lg(),
        "Reading the rows of one FX forward publish.");
}
