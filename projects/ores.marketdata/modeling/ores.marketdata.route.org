A route names the one source that serves a party for a provider. A party has at most one
route for each provider, so it reads one value of each instrument and the mapping and the series
key stay unambiguous. A source switch is one new route and nothing else.

The source must belong to the provider of the route, and the party must hold a subscription to
the provider. A change of route is a new version with a reason and a commentary, and the history
keeps the earlier routes. The source must also be ready; the adapter state that says so is held by
the task that tracks adapters.

#+begin_src sql :name route_checks :implements d4e8e136-828d-440a-ba59-82ba2b9d3e2e
    -- The party holds a subscription to the provider the route serves.
    if not exists (
        select 1 from ores_marketdata_subscriptions_tbl
        where tenant_id = NEW.tenant_id
          and party_id = NEW.party_id
          and provider_code = NEW.provider_code
          and valid_to = ores_utility_infinity_timestamp_fn()
    ) then
        raise exception 'Invalid route: party % holds no subscription to provider %.',
            NEW.party_id, NEW.provider_code
            using errcode = '23503';
    end if;

    -- The source belongs to the provider of the route. Sources are rows of the
    -- system tenant, which every tenant reads.
    if not exists (
        select 1 from ores_marketdata_feeds_tbl
        where tenant_id in (NEW.tenant_id, ores_utility_system_tenant_id_fn())
          and id = NEW.feed_id
          and publisher_code = NEW.provider_code
          and valid_to = ores_utility_infinity_timestamp_fn()
    ) then
        raise exception 'Invalid route: source % is not a source of provider %.',
            NEW.feed_id, NEW.provider_code
            using errcode = '23514';
    end if;

*** read_latest_all_tenants
:PROPERTIES:
:ID: 5C2E8A14-7D3B-4F96-A1E5-9B4D6C0F3E27
:END:

Reads the latest routes across all tenants. The ingest loop resolves the parties routed
to a source regardless of tenant, so it cannot filter on one. Queries only on
=valid_to == max=.

#+begin_src cpp :name implementation :implements F2EB1914-5E94-42CA-9C77-46BDB364BF9E
std::vector<domain::route>
route_repository::read_latest_all_tenants(context ctx) {
    static constexpr auto max = db_timestamp::max();
    const auto query = query_builder::read<std::vector<route_entity>> |
                       query_builder::where("valid_to"_c == max) |
                       query_builder::order_by("id"_c);

    return query_builder::execute_read_query<route_entity, domain::route>(
        ctx,
        query,
        [](const auto& entities) { return route_mapper::map(entities); },
        lg(),
        "Reading latest routes across all tenants");
}
