The bodies the conventions service compiles instead of writing one call per
family by hand: the staged-import dispatcher, which decodes one family's
payload into its own type, and the payload reader, which serialises one
family's rows the way the wire carries them. No instantiation reflects more
than one family.

The fragment is an include, not a translation unit. It is compiled where the
row helpers (=replace_party_rows=, =replace_rows_by_id=,
=add_missing_world_rows=), the repositories and the stamp helper are visible,
which is why the includes it needs travel as a second fragment beside it.

The output is an =.inc= file, so clang-format does not run over it and the
template states the layout: the continuation columns, the blank line between
bodies, and the one-call-per-family grouping the reader scans.

variables and their semantics.

// Staged families: each branch decodes one family's payload into its own
// type, so no instantiation reflects more than one family.
ORES_REFDATA_CORE_EXPORT bool apply_staged_family(const ores::database::context& ctx,
                                                  const std::string& family,
                                                  const std::string& payload,
                                                  conventions_apply_result& result) {
{{#families}}
    if (family == "{{key}}") {
        const auto decoded =
            rfl::json::read<{{domain_type}}>(payload);
        if (!decoded)
            throw std::runtime_error(
                "Staged conventions payload for family "
                "\"{{key}}\" does not decode.");
        auto rows = std::vector<{{domain_type}}>{
            std::move(*decoded)};
        ores::service::messaging::stamp(rows.front(), ctx);
{{#is_world}}
        add_missing_world_rows(ctx,
                               {{entity}}_repository(),
                               std::move(rows),
                               result.world_kept);
{{/is_world}}
{{^is_world}}
        {{#is_party}}replace_party_rows{{/is_party}}{{#is_by_id}}replace_rows_by_id{{/is_by_id}}(ctx,
                           {{entity}}_repository(),
                           std::move(rows));
{{/is_world}}
        return true;
    }
{{/families}}
    return false;
}

// Payload reads: each branch reads one family's live rows and serialises each
// as its own type, so an in-process caller sees the bytes the wire carries and
// no branch touches another family's type.
ORES_REFDATA_CORE_EXPORT std::vector<std::string>
read_family_payloads(const ores::database::context& ctx, const std::string& family) {
{{#families}}
    if (family == "{{key}}") {
        std::vector<std::string> payloads;
        for (const auto& row : {{entity}}_repository().read_latest(ctx))
            payloads.push_back(rfl::json::write(row));
        return payloads;
    }
{{/families}}
    throw std::logic_error("The registry does not know the convention family '" + family + "'.");
}
{{/convention_registry}}
