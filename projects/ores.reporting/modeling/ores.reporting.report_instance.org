A single execution of a report_definition. Created automatically when the
scheduler fires a trigger for an active definition.

Lifecycle is managed through the report_instance_lifecycle FSM machine.
fsm_state_id points to the current state in ores_dq_fsm_states_tbl.

started_at is NULL when the instance is cancelled or skipped before execution
begins. completed_at is NULL while running or in a terminal-before-start state.

*** read_latest_by_definition_ids

Reads the run that started last, for each of the definitions named. This cannot
be templated: the generated list offers offset and limit over the rows the
policy admits, so a caller that wanted one row per definition would page the
party's whole run history and reduce it in C++. The store answers the question
in one statement instead, and the reduction never leaves the database.

The rows themselves are read back through the generated =read_latest(context,
ids)=, so the entity, the mapper and the tenant and validity conditions are the
same code the generated reads use.

     * parties are not the caller's to read: the store's policy decides that,
     * not this method.

#+begin_src cpp :name implementation :implements F2EB1914-5E94-42CA-9C77-46BDB364BF9E
std::vector<domain::report_instance>
report_instance_repository::read_latest_by_definition_ids(
    context ctx, const std::vector<boost::uuids::uuid>& definition_ids) {
    if (definition_ids.empty())
        return {};

    BOOST_LOG_SEV(lg(), debug) << "Reading the latest instance of "
                               << definition_ids.size() << " definition(s).";

    // One bound array rather than one parameter per id, so a party with a
    // hundred definitions is still one statement.
    std::string id_array("{");
    for (std::size_t i = 0; i < definition_ids.size(); ++i) {
        if (i > 0)
            id_array += ',';
        id_array += boost::uuids::to_string(definition_ids[i]);
    }
    id_array += '}';

    // The run that started last is the greatest started_at. A run cancelled or
    // skipped before it started has no started_at, so valid_from decides
    // between those, and NULLS LAST keeps a started run ahead of one that
    // never started. The tenant and the policy on the table narrow the scan
    // exactly as they do for the generated reads; no party is stated here.
    const auto rows = ores::database::repository::execute_parameterized_string_query(
        ctx,
        "SELECT DISTINCT ON (definition_id) id::text"
        " FROM ores_reporting_report_instances_tbl"
        " WHERE tenant_id = $1::uuid"
        "   AND valid_to = ores_utility_infinity_timestamp_fn()"
        "   AND definition_id = ANY($2::uuid[])"
        " ORDER BY definition_id, started_at DESC NULLS LAST, valid_from DESC",
        {ctx.tenant_id().to_string(), id_array},
        lg(),
        "Reading the latest instance of each definition");

    if (rows.empty())
        return {};
    return read_latest(ctx, rows);
}

*** read_health_facts_by_instance_ids

Reads what a run's health is derived from: the name of its lifecycle state and
the number of errors it recorded. The health word itself is the service's to
state, because the mapping from a state to a word is a decision about the
screen's vocabulary and not a fact the store holds.

Two reads, not one, because the lifecycle states and the runs live on opposite
sides of the tenant boundary. The states are the system tenant's reference data
and the row-level policy on their table admits them only to a session that
states the system tenant; the runs are the caller's and are read under the
caller's tenant. A single join would therefore find no state for every tenant
but the system one, which is why the states are resolved by id in their own
read, under the tenant that owns them. The reporting service holds a SELECT
grant on =ores_dq_fsm_= for exactly this read, and reading system reference data
under the system tenant is what the service's own state lookups already do.

#+begin_src cpp :name implementation :implements F2EB1914-5E94-42CA-9C77-46BDB364BF9E
std::vector<report_instance_health_facts>
report_instance_repository::read_health_facts_by_instance_ids(
    context ctx, const std::vector<boost::uuids::uuid>& instance_ids) {
    if (instance_ids.empty())
        return {};

    BOOST_LOG_SEV(lg(), debug) << "Reading the health facts of " << instance_ids.size()
                               << " run(s).";

    std::string id_array("{");
    for (std::size_t i = 0; i < instance_ids.size(); ++i) {
        if (i > 0)
            id_array += ',';
        id_array += boost::uuids::to_string(instance_ids[i]);
    }
    id_array += '}';

    // The runs and their errors are the caller's, so they are read under the
    // caller's tenant and the policy on each table decides what the session
    // may see. No party is stated here.
    const auto rows = ores::database::repository::execute_parameterized_multi_column_query(
        ctx,
        "SELECT ri.id::text, coalesce(ri.fsm_state_id::text, ''), count(e.id)::text"
        " FROM ores_reporting_report_instances_tbl ri"
        " LEFT JOIN ores_reporting_report_error_events_tbl e"
        "   ON e.tenant_id = ri.tenant_id"
        "  AND e.report_instance_id = ri.id"
        "  AND e.severity = 'error'"
        " WHERE ri.tenant_id = $1::uuid"
        "   AND ri.valid_to = ores_utility_infinity_timestamp_fn()"
        "   AND ri.id = ANY($2::uuid[])"
        " GROUP BY ri.id, ri.fsm_state_id",
        {ctx.tenant_id().to_string(), id_array},
        lg(),
        "Reading the health facts of each run");

    std::vector<report_instance_health_facts> facts;
    std::vector<std::string> state_id_of;
    state_id_of.reserve(rows.size());
    std::string state_id_array("{");
    for (const auto& row : rows) {
        if (row.size() < 3 || !row[0] || !row[1] || !row[2])
            continue;
        report_instance_health_facts entry;
        entry.report_instance_id = boost::lexical_cast<boost::uuids::uuid>(*row[0]);
        entry.error_count = std::stoll(*row[2]);
        facts.push_back(std::move(entry));
        state_id_of.push_back(*row[1]);
        if (row[1]->empty())
            continue;
        if (state_id_array.size() > 1)
            state_id_array += ',';
        state_id_array += *row[1];
    }
    state_id_array += '}';

    if (facts.empty() || state_id_array == "{}")
        return facts;

    // The states are the system tenant's reference data, so they are read under
    // that tenant. The caller's tenant cannot see them at all, which is why
    // this read exists rather than the screen resolving a state id itself.
    const auto state_names = ores::database::repository::execute_parameterized_multi_column_query(
        ctx.with_tenant(ores::utility::uuid::tenant_id::system(), ctx.actor()),
        "SELECT id::text, name"
        " FROM ores_dq_fsm_states_tbl"
        " WHERE id = ANY($1::uuid[])"
        "   AND valid_to = ores_utility_infinity_timestamp_fn()",
        {state_id_array},
        lg(),
        "Reading the lifecycle states of each run");

    std::map<std::string, std::string> name_of;
    for (const auto& row : state_names) {
        if (row.size() >= 2 && row[0] && row[1])
            name_of.emplace(*row[0], *row[1]);
    }

    for (std::size_t i = 0; i < facts.size(); ++i) {
        if (state_id_of[i].empty())
            continue;
        if (const auto it = name_of.find(state_id_of[i]); it != name_of.end())
            facts[i].state_name = it->second;
    }
    return facts;
}

#+begin_src cpp :name implementation :implements D3E7A5F2-8B4C-4D1E-9A6F-2E1C7B4D8A3F
messaging::count_report_definitions_by_health_response
report_instance_service::count_report_definitions_by_health(
    const messaging::count_report_definitions_by_health_request& request) {
    messaging::count_report_definitions_by_health_response response;
    // The cap is the same one the generated filter states.
    if (request.search && request.search->size() > 256) {
        response.result =
            refuse(outcome_code::filter_too_large, {.field = "search", .limit = "256"});
        return response;
    }

    // The words in the order the screen draws them, with the empty word last.
    constexpr std::array<std::string_view, 6> words{
        "failed", "completed with errors", "running", "never run", "completed clean", ""};
    std::map<std::string, std::int64_t> tally;
    for (const auto word : words)
        tally.emplace(std::string(word), 0);
    response.total = 0;

    messaging::list_report_definitions_request list;
    list.limit = 1000;
    list.filter = messaging::report_definitions_filter{};
    list.filter->party_id = request.party_id;
    list.filter->search = request.search;

    report_definition_service definitions(ctx_);
    for (;;) {
        const auto page = definitions.list_report_definitions(list);
        if (page.result.outcome != ores::utility::domain::outcome::ok) {
            response.result = page.result;
            return response;
        }

        std::vector<boost::uuids::uuid> definition_ids;
        definition_ids.reserve(page.definitions.size());
        for (const auto& definition : page.definitions)
            definition_ids.push_back(definition.id);

        // One latest run per definition that has run, and the health facts of
        // those runs, so a page costs two reads however many runs exist.
        std::map<boost::uuids::uuid, boost::uuids::uuid> run_of;
        std::vector<boost::uuids::uuid> run_ids;
        for (const auto& run : repo_.read_latest_by_definition_ids(ctx_, definition_ids)) {
            run_of.emplace(run.definition_id, run.id);
            run_ids.push_back(run.id);
        }
        std::map<boost::uuids::uuid, std::string> word_of;
        for (const auto& facts : repo_.read_health_facts_by_instance_ids(ctx_, run_ids))
            word_of.emplace(facts.report_instance_id,
                            health_word(facts.state_name, facts.error_count));

        for (const auto& id : definition_ids) {
            const auto run = run_of.find(id);
            if (run == run_of.end()) {
                ++tally.at("never run");
                continue;
            }
            // A run the session cannot read states no word, so it is counted
            // with the runs whose state has none.
            const auto word = word_of.find(run->second);
            ++tally.at(word == word_of.end() ? std::string() : word->second);
        }
        response.total += static_cast<std::int64_t>(definition_ids.size());

        list.offset += list.limit;
        if (page.definitions.size() < list.limit)
            break;
    }

    for (const auto word : words) {
        messaging::report_health_count entry;
        entry.health = std::string(word);
        entry.count = tally.at(entry.health);
        response.counts.push_back(std::move(entry));
    }
    return response;
}
