"""Plant one violation for each gate compass-ng answers for.

Run from the repository root on a throwaway branch. Each plant names the gate
it must fail; the run that follows proves which gates caught theirs.
"""
from pathlib import Path

PLANTS = []


def plant(gate):
    def register(fn):
        PLANTS.append((gate, fn))
        return fn
    return register


def append(path, text):
    p = Path(path)
    p.write_text(p.read_text() + text)


def replace(path, old, new):
    p = Path(path)
    t = p.read_text()
    assert t.count(old) >= 1, (path, old)
    p.write_text(t.replace(old, new, 1))


def write(path, text):
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(text)


def inside_markers(path, text):
    p = Path(path)
    lines = p.read_text().splitlines(keepends=True)
    at = next(i for i, line in enumerate(lines) if "BEGIN generated" in line)
    lines.insert(at + 1, text)
    p.write_text("".join(lines))


def table_cell(path):
    p = Path(path)
    lines = p.read_text().splitlines(keepends=True)
    sep = next(i for i, line in enumerate(lines) if line.startswith("|-"))
    row = lines[sep + 1]
    lines[sep + 1] = row.replace("| ", "| planted ", 1)
    p.write_text("".join(lines))

@plant("rule-identifiers")
def _():
    for path in sorted(Path("doc").rglob("*.org")):
        lines = _lines(path)
        hits = [i for i, l in enumerate(lines) if _re.match(r"- \*[A-Z]+(-[A-Z]+)*-\d\d\*", l)]
        if hits:
            lines.insert(hits[0] + 1, lines[hits[0]])
            path.write_text("".join(lines))
            return
    raise AssertionError("no keyed rule found")
