+/**
+ * A refused read is a question the server will not answer, such as a currency it
+ * holds no ladder for. The server's words are the reason, so they travel as they
+ * are and the screen shows them.
+ */
+function refused(message: string): HttpFailure {
+    return new HttpFailure(422, { code: 'invalid-request', message: message });
+}
+
+/**
+ * The reply, once the server has agreed to answer it.
+ *
+ * A refusal is the server's, so its words travel as the reason. The schema is
+ * what makes the rest of the answer safe to read: a field the server did not
+ * send is its default here and nowhere else.
+ */
+async function ask<Schema extends z.ZodType>(
+    session: LiveSession,
+    subject: string,
+    body: Record<string, unknown>,
+    schema: Schema,
+): Promise<z.infer<Schema>> {
+    const reply = (await session.client.callAuthenticated(
+        subject,
+        body,
+        schema,
+    )) as z.infer<Schema> & {
+        success?: boolean;
+        message?: string;
+    };
+    if (reply.success === false) {
+        throw refused(reply.message ?? '');
+    }
+    return reply;
+}

+export function registerMarketReadRoutes(
+    server: FastifyInstance,
+    requireSession: (request: FastifyRequest) => LiveSession,
+): void {
+    // The catalogue of the session party: one entry per instrument, with its
+    // axes in draw order and the verdict on each source.
+    server.post('/api/market/instruments', async (request) => {
+        const session = requireSession(request);
+        const body = input(instrumentsBody, request.body);
+        const reply = await ask(
+            session,
+            subjects.list_instruments_request,
+            {
+                party_id: '',
+                query: body.query,
+                kinds: body.kinds,
+                provider_code: body.provider_code,
+                offset: body.offset,
+                limit: body.limit,
+            },
+            instrumentsReply,
+        );
+        return {
+            instruments: reply.instruments,
+            total: reply.total,
+            truncated: reply.truncated,
+        };
+    });

+function oneOf<T extends string>(value: string | null, allowed: readonly T[], fallback: T): T {
+    return allowed.find((candidate) => candidate === value) ?? fallback;

+            ? percent(point.zero_rate)
+            : metric === 'forward'
+              ? percent(point.implied_forward)
+              : point.discount_factor,
+    );
+    const series: ChartSeries[] = [
+        { points: curve, colour: SERIES_COLOURS[0], fill: metric !== 'discount', strong: true },
+    ];
+    if (metric !== 'discount') {
+        // The market's own quote at each pillar, matched by tenor so the two lines
+        // share an x. A pillar the market did not quote is a gap, not a zero.
+        const market = new Map(
+            data.market_pillars.map((pillar) => [pillar.tenor, percent(pillar.rate)]),
+        );
+        series.push({
+            points: data.bootstrapped.map((point) => market.get(point.tenor) ?? null),
+            colour: SERIES_COLOURS[2],
+            dashed: true,
+            dots: true,
+        });
+    }
+    return series;

+ * server facts. The screen adds only what the prototype asked for and no read
+ * states: the best quote of a rung and its distance in basis points from the
+ * consensus.
+ */
+
+const CURRENCIES = ['USD', 'EUR', 'GBP', 'JPY', 'CHF', 'AUD', 'CAD', 'NZD'] as const;
+
+/// The forward start a ladder is quoted from: spot, or a broken date.
+const STARTS = ['0D', '3M', '1Y'] as const;
+
+/// How much of the curve the ladder shows, by the year fraction the server states.
+const WINDOWS = ['short', 'all', 'long'] as const;
+
+type Window = (typeof WINDOWS)[number];
+
+function oneOf<T extends string>(value: string | null, allowed: readonly T[], fallback: T): T {
+    return allowed.find((candidate) => candidate === value) ?? fallback;
+}
+
+/** Whether a rung falls in the window, by the year fraction the server dated it with. */
+function inWindow(rung: RateLadderRung, window: Window): boolean {
+    if (window === 'short') return rung.year_fraction < 1.0 / 12.0;
+    if (window === 'long') return rung.year_fraction >= 1.0 && rung.year_fraction <= 10.0;
+    return true;
+}
+
+function asOfInstant(date: string): string {
+    return `${date} 23:59:59Z`;
+}
+
+function todayIso(): string {
+    return new Date().toISOString().slice(0, 10);
+}
+
+function percent(value: number, language: string): string {
+    return (value * 100).toLocaleString(language, {
+        minimumFractionDigits: 4,
+        maximumFractionDigits: 4,
+    });
+}
+
+function distance(best: number | undefined, consensus: number): string {
+    if (best === undefined) return '—';
+    const bp = (best - consensus) * 10000;
+    return `${bp >= 0 ? '+' : ''}${bp.toFixed(1)} bp`;
+}
+
+function ageOf(rung: RateLadderRung, language: string): string {
+    const newest = [...rung.quotes].sort((a, b) => a.age_seconds - b.age_seconds)[0];
+    if (newest === undefined) return '—';
+    const now = new Date();
+    return relativeTime(new Date(now.getTime() - newest.age_seconds * 1000), now, language);
+}

+                forwardStart: start,

+        retry: false,
+    });
+
+    const rows = (ladder.data?.rungs ?? []).filter((rung) => inWindow(rung, window));
+    // One column per provider, in the order the server lists them, so a gap on a
+    // rung is a gap in a column a person can see.
+    const quoted = [
